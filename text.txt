# Install GPIO library for Raspberry Pi
sudo apt install python3-rpi.gpio -y

# Install Python3 and pip (if not already installed)
sudo apt install python3 python3-pip -y

# Install OpenCV (cv2) with Raspberry Pi optimizations
sudo apt install python3-opencv -y

# Install PyTorch for ARM (Raspberry Pi-compatible version)
pip3 install torch torchvision --extra-index-url https://download.pytorch.org/whl/arm

pip install ultralytics  # Installs YOLOv8
pip install opencv-python-headless  # Lightweight OpenCV for Pi

pip install git+https://github.com/seveas/python-prctl.git

pip3 list | grep picamera

dpkg -l | grep raspberrypi-ui-mods

ps aux | grep chromium

rm -f ~/.config/chromium/SingletonLock

rm -f ~/.config/chromium/SingletonCookie

git clone --recursive https://github.com/pytorch/pytorch
cd pytorch

export USE_NINJA=OFF
export USE_MKLDNN=OFF

pip3 install -r requirements.txt

python3 setup.py install

# Install dependencies
sudo apt update
sudo apt install python3.9 python3.9-venv python3.9-dev -y

# Create and activate virtual environment
python3.9 -m venv torch_env
source torch_env/bin/activate

# Download a compatible PyTorch wheel
wget https://github.com/nmilosev/pytorch-arm-builds/releases/download/v1.13.0/torch-1.13.0-cp39-cp39-linux_aarch64.whl

# Install it
pip install torch-1.13.0-cp39-cp39-linux_aarch64.whl

# Then install Ultralytics
pip install ultralytics==8.2.0
